\section*{Questões e Respostas}

\begin{enumerate}[leftmargin=*, itemsep=15pt]

\item Se 5 cadernos custam R\$ 40, quanto custarão 8 cadernos? \\
\textbf{Resposta: R\$ 64,00} \\
Diretamente proporcional: mais cadernos = mais dinheiro \\
$\frac{5}{8} = \frac{40}{x} \rightarrow 5x = 320 \rightarrow x = 64$

\item Uma máquina produz 300 peças em 6 horas. Quantas peças produzirá em 10 horas? \\
\textbf{Resposta: 500 peças} \\
Diretamente proporcional: mais horas = mais peças \\
$\frac{6}{10} = \frac{300}{x} \rightarrow 6x = 3000 \rightarrow x = 500$

\item Se 12 litros de combustível permitem percorrer 180 km, quantos quilômetros serão percorridos com 20 litros? \\
\textbf{Resposta: 300 km} \\
Diretamente proporcional: mais combustível = mais km \\
$\frac{12}{20} = \frac{180}{x} \rightarrow 12x = 3600 \rightarrow x = 300$

\item Um pedreiro recebe R\$ 150 por dia. Quanto receberá em 18 dias? \\
\textbf{Resposta: R\$ 2.700,00} \\
Diretamente proporcional: mais dias = mais dinheiro \\
$\frac{1}{18} = \frac{150}{x} \rightarrow x = 150 \times 18 = 2700$

\item 4 torneiras enchem um reservatório em 2 horas. Quantas torneiras serão necessárias para encher 3 reservatórios no mesmo tempo? \\
\textbf{Resposta: 12 torneiras} \\
Diretamente proporcional: mais reservatórios = mais torneiras \\
4 torneiras $\rightarrow$ 1 reservatório \\
$x$ torneiras $\rightarrow$ 3 reservatórios \\
$x = 4 \times 3 = 12$

\item 7 operários constroem um muro em 5 dias. Quantos operários são necessários para construir 3 muros no mesmo período? \\
\textbf{Resposta: 21 operários} \\
Diretamente proporcional: mais muros = mais operários \\
7 operários $\rightarrow$ 1 muro \\
$x$ operários $\rightarrow$ 3 muros \\
$x = 7 \times 3 = 21$

\item Uma receita utiliza 250 g de farinha para fazer um bolo. Quantos gramas serão necessários para 6 bolos? \\
\textbf{Resposta: 1.500 g} \\
Diretamente proporcional: mais bolos = mais farinha \\
$\frac{1}{6} = \frac{250}{x} \rightarrow x = 250 \times 6 = 1500$

\item 3 kg de carne servem 6 pessoas. Quantos kg serão necessários para 15 pessoas? \\
\textbf{Resposta: 7,5 kg} \\
Diretamente proporcional: mais pessoas = mais carne \\
$\frac{6}{15} = \frac{3}{x} \rightarrow 6x = 45 \rightarrow x = 7,5$

\item Se 8 metros de tecido custam R\$ 200, quanto custarão 15 metros? \\
\textbf{Resposta: R\$ 375,00} \\
Diretamente proporcional: mais metros = mais dinheiro \\
$\frac{8}{15} = \frac{200}{x} \rightarrow 8x = 3000 \rightarrow x = 375$

\item Uma gráfica imprime 2.400 panfletos em 3 horas. Quantos panfletos imprimirá em 8 horas? \\
\textbf{Resposta: 6.400 panfletos} \\
Diretamente proporcional: mais horas = mais panfletos \\
$\frac{3}{8} = \frac{2400}{x} \rightarrow 3x = 19200 \rightarrow x = 6400$

\item Se 6 livros custam R\$ 90, quanto custarão 14 livros? \\
\textbf{Resposta: R\$ 210,00} \\
Diretamente proporcional: mais livros = mais dinheiro \\
$\frac{6}{14} = \frac{90}{x} \rightarrow 6x = 1260 \rightarrow x = 210$

\item Um carro percorre 150 km com 10 litros de combustível. Quantos quilômetros percorrerá com 25 litros? \\
\textbf{Resposta: 375 km} \\
Diretamente proporcional: mais combustível = mais km \\
$\frac{10}{25} = \frac{150}{x} \rightarrow 10x = 3750 \rightarrow x = 375$

\item 9 trabalhadores constroem uma cerca em 4 dias. Quantos trabalhadores serão necessários para construir 3 cercas no mesmo período? \\
\textbf{Resposta: 27 trabalhadores} \\
Diretamente proporcional: mais cercas = mais trabalhadores \\
9 trabalhadores $\rightarrow$ 1 cerca \\
$x$ trabalhadores $\rightarrow$ 3 cercas \\
$x = 9 \times 3 = 27$

\item Uma fábrica produz 500 garrafas em 5 horas. Quantas garrafas produzirá em 12 horas? \\
\textbf{Resposta: 1.200 garrafas} \\
Diretamente proporcional: mais horas = mais garrafas \\
$\frac{5}{12} = \frac{500}{x} \rightarrow 5x = 6000 \rightarrow x = 1200$

\item Se 4 kg de arroz custam R\$ 32, quanto custarão 10 kg? \\
\textbf{Resposta: R\$ 80,00} \\
Diretamente proporcional: mais kg = mais dinheiro \\
$\frac{4}{10} = \frac{32}{x} \rightarrow 4x = 320 \rightarrow x = 80$

\item Uma impressora imprime 120 páginas em 6 minutos. Quantas páginas imprimirá em 15 minutos? \\
\textbf{Resposta: 300 páginas} \\
Diretamente proporcional: mais minutos = mais páginas \\
$\frac{6}{15} = \frac{120}{x} \rightarrow 6x = 1800 \rightarrow x = 300$

\item 2 caixas contêm 48 lápis. Quantos lápis haverá em 7 caixas iguais? \\
\textbf{Resposta: 168 lápis} \\
Diretamente proporcional: mais caixas = mais lápis \\
$\frac{2}{7} = \frac{48}{x} \rightarrow 2x = 336 \rightarrow x = 168$

\item Um caminhão transporta 8 toneladas de areia em uma viagem. Quantas toneladas transportará em 15 viagens? \\
\textbf{Resposta: 120 toneladas} \\
Diretamente proporcional: mais viagens = mais toneladas \\
$\frac{1}{15} = \frac{8}{x} \rightarrow x = 8 \times 15 = 120$

\item 5 metros de fio pesam 200 g. Quanto pesarão 18 metros do mesmo fio? \\
\textbf{Resposta: 720 g} \\
Diretamente proporcional: mais metros = mais peso \\
$\frac{5}{18} = \frac{200}{x} \rightarrow 5x = 3600 \rightarrow x = 720$

\item Se 3 litros de tinta pintam 24 m² de parede, quantos metros quadrados serão pintados com 10 litros? \\
\textbf{Resposta: 80 m²} \\
Diretamente proporcional: mais tinta = mais área \\
$\frac{3}{10} = \frac{24}{x} \rightarrow 3x = 240 \rightarrow x = 80$

\item Três trabalhadores fazem um serviço em 12 dias. Quantos dias levariam 6 trabalhadores para fazer o mesmo serviço? \\
\textbf{Resposta: 6 dias} \\
Inversamente proporcional: mais trabalhadores = menos dias \\
$3 \times 12 = 6 \times x \rightarrow 36 = 6x \rightarrow x = 6$

\item Se 4 máquinas fazem um trabalho em 5 horas, quantas horas seriam necessárias para que 10 máquinas realizassem o mesmo trabalho? \\
\textbf{Resposta: 2 horas} \\
Inversamente proporcional: mais máquinas = menos horas \\
$4 \times 5 = 10 \times x \rightarrow 20 = 10x \rightarrow x = 2$

\item Dez operários fazem um trabalho em 8 dias. Quantos operários são necessários para que termine em 4 dias? \\
\textbf{Resposta: 20 operários} \\
Inversamente proporcional: mais operários = menos dias \\
$10 \times 8 = x \times 4 \rightarrow 80 = 4x \rightarrow x = 20$

\item Uma torneira enche um reservatório em 10 horas. Se abrirmos mais 2 torneiras iguais, quanto tempo será preciso? \\
\textbf{Resposta: 3,33 horas ou 3h20min} \\
Inversamente proporcional: mais torneiras = menos tempo \\
$1 \times 10 = 3 \times x \rightarrow 10 = 3x \rightarrow x \approx 3,33$

\item Um grupo de 8 pessoas leva 4 dias para pintar uma casa. Quantos dias levará um grupo de 12 pessoas? \\
\textbf{Resposta: 2,67 dias ou 2 dias e 16h} \\
Inversamente proporcional: mais pessoas = menos dias \\
$8 \times 4 = 12 \times x \rightarrow 32 = 12x \rightarrow x \approx 2,67$

\item Uma equipe de 5 jardineiros leva 15 horas para terminar um jardim. Quantas horas levariam 10 jardineiros? \\
\textbf{Resposta: 7,5 horas} \\
Inversamente proporcional: mais jardineiros = menos horas \\
$5 \times 15 = 10 \times x \rightarrow 75 = 10x \rightarrow x = 7,5$

\item Um serviço é feito por 9 trabalhadores em 20 dias. Quantos trabalhadores são necessários para que o serviço seja feito em 15 dias? \\
\textbf{Resposta: 12 trabalhadores} \\
Inversamente proporcional: menos dias = mais trabalhadores \\
$9 \times 20 = x \times 15 \rightarrow 180 = 15x \rightarrow x = 12$

\item 4 carros levam 18 horas para completar um trabalho. Quantos carros seriam necessários para completar em 6 horas? \\
\textbf{Resposta: 12 carros} \\
Inversamente proporcional: menos horas = mais carros \\
$4 \times 18 = x \times 6 \rightarrow 72 = 6x \rightarrow x = 12$

\item Um encanador leva 9 horas para consertar 3 canos. Quantas horas levaria para consertar 6 canos? \\
\textbf{Resposta: 18 horas} \\
Diretamente proporcional: mais canos = mais horas \\
$\frac{3}{6} = \frac{9}{x} \rightarrow 3x = 54 \rightarrow x = 18$

\item Se 15 pintores fazem um trabalho em 5 dias, quantos dias levariam 30 pintores? \\
\textbf{Resposta: 2,5 dias} \\
Inversamente proporcional: mais pintores = menos dias \\
$15 \times 5 = 30 \times x \rightarrow 75 = 30x \rightarrow x = 2,5$

\end{enumerate}
