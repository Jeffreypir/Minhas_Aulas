\textbf{Princípio Fundamental da Contagem}  
\[
N = n_1 \cdot n_2 \cdot \dots \cdot n_k
\]

\vspace{0.2cm}

\textbf{Arranjo de $n$ elementos tomados $p$ a $p$}  
\[
A_n^p = \frac{n!}{(n-p)!}
\]

\vspace{0.2cm}

\textbf{Permutação de $n$ elementos}  
\[
P_n = n!
\]

\vspace{0.2cm}

\textbf{Permutação com elementos repetidos}  
\[
P = \frac{n!}{n_1! \, n_2! \dots n_k!}
\]

\vspace{0.2cm}

\textbf{Combinação de $n$ elementos tomados $p$ a $p$}  
\[
C_n^p = \binom{n}{p} = \frac{n!}{p!(n-p)!}
\]
