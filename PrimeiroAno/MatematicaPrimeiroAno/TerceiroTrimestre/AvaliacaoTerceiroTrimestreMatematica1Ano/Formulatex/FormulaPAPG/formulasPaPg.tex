\textbf{Média:}

\begin{align*}
\bar{x} &= \frac{x_1 + x_2 + \dots + x_n}{n} \\
        &= \frac{\sum_{i=1}^{n} x_i}{n}
\end{align*}

\textbf{Mediana:}

\begin{align*}
\text{Se $n$ ímpar:} \quad & \text{Mediana} = x_{\frac{n+1}{2}} \\
\text{Se $n$ par:} \quad & \text{Mediana} = \frac{x_{\frac{n}{2}} + x_{\frac{n}{2}+1}}{2}
\end{align*}

\textbf{Moda:}

\begin{align*}
\text{Moda} &= \text{valor que ocorre com maior frequência nos dados}
\end{align*}
