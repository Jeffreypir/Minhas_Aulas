\textbf{Média:}
\begin{align*}
Média &= \frac{x_1 + x_2 + \dots + x_n}{n} \\
\end{align*}

\textbf{Mediana:}
\begin{align*}
    &\text{Se $n$ ímpar:} \quad  \text{Mediana} = x_{\left( \frac{n+1}{2}\right )} \\
    &\text{Se $n$ par:} \quad  \text{Mediana} = \frac{x_{\left (\frac{n}{2}\right)} + x_{\left(\frac{n}{2}+1}\right)}{2}
\end{align*}

\textbf{Moda:}
\begin{align*}
\text{Moda} &= \text{valor que ocorre com maior frequência nos dados}
\end{align*}
